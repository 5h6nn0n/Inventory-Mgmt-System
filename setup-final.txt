DROP TABLE FN24_ORDER_LINE CASCADE CONSTRAINTS;
DROP TABLE FN24_ORDER CASCADE CONSTRAINTS;
DROP TABLE FN24_PASTRY CASCADE CONSTRAINTS;
DROP TABLE FN24_CUSTOMER CASCADE CONSTRAINTS;

 

CREATE TABLE FN24_CUSTOMER
(
	CUST_CODE		NUMBER(10)			,	
	CUST_BUSINESS_NAME	VARCHAR2(40)	NOT NULL	,
	CUST_AREA_CODE		NUMBER(3)	NOT NULL	,
	CUST_PHONE_NUM		NUMBER(7)	NOT NULL	,
	CONSTRAINT FN24_CSTMR_CST_CD_PK PRIMARY KEY (CUST_CODE)
);



CREATE TABLE FN24_PASTRY
(
	PASTRY_CODE		NUMBER(10)			,
	PASTRY_NAME		VARCHAR2(40)	NOT NULL	,			
	PASTRY_PRICE_PER_UNIT	NUMBER(12,2)    NOT NULL        , 
	PASTRY_BAKED_DATE	DATE		NOT NULL	,
	PASTRY_QTY_ON_HAND   NUMBER(10) ,
	CONSTRAINT FN24_PASTRY_PSTRY_CD_PK PRIMARY KEY (PASTRY_CODE)
);

CREATE TABLE FN24_ORDER
(
	ORDER_NUM		NUMBER(10)			,	
	ORDER_DATE		DATE		NOT NULL	,
	CUST_CODE		NUMBER(10)	NOT NULL	,
	CONSTRAINT FN24_ORDR_ORDR_NUM_PK PRIMARY KEY (ORDER_NUM)	, 
	CONSTRAINT FN24_ORDR_CUST_CODE_FK FOREIGN KEY (CUST_CODE) REFERENCES FN24_CUSTOMER(CUST_CODE)
);

CREATE TABLE FN24_ORDER_LINE
(
	ORDER_NUM		NUMBER(10)			,
	ORDER_LINE_NUM		NUMBER(10)	 		,
	PASTRY_CODE		NUMBER(10)	NOT NULL	,
	QUANTITY_ORDERED	NUMBER(10)	NOT NULL    	,
	CONSTRAINT FN24_ORDER_NUM_LINE_PK PRIMARY KEY (ORDER_NUM,ORDER_LINE_NUM),
	CONSTRAINT FN24_ORDR_NUM_FK FOREIGN KEY (ORDER_NUM) REFERENCES FN24_ORDER (ORDER_NUM), 
	CONSTRAINT FN24_ORDR_LNE_PASTRY_CD_FK FOREIGN KEY (PASTRY_CODE) REFERENCES FN24_PASTRY (PASTRY_CODE) 
);


/*******   Create sequence *******/

DROP SEQUENCE FN24_CUSTOMER_PK_SEQ;

 

CREATE SEQUENCE FN24_CUSTOMER_PK_SEQ
START WITH 200
INCREMENT BY 1
NOCYCLE
NOCACHE;
